The script must:
1. Track PASS/FAIL counts and print a summary at the end
2. Use a helper function `assert_ok` that checks exit codes and prints test results
3. Use python3 for JSON parsing (no jq dependency)
4. Follow user journey order per D-01: clean state -> init -> usage -> scan -> apply -> status -> uninstall

Structure the script with these sections:

```bash
#!/usr/bin/env bash
set -euo pipefail

# Configuration
CLI="node $(pwd)/dist/cli.js"
SETTINGS="$HOME/.claude/settings.json"
COMMANDS_DIR="$HOME/.claude/commands/evolve"
DATA_DIR="$HOME/.harness-evolve"
PASS=0
FAIL=0

# Helper functions
pass() { echo "  PASS: $1"; ((PASS++)); }
fail() { echo "  FAIL: $1 -- $2"; ((FAIL++)); }

# Wave 0: Clean State
echo "=== Wave 0: Clean State Setup ==="
# Backup existing settings.json
cp "$SETTINGS" "$SETTINGS.test-backup" 2>/dev/null || true
# Full uninstall + purge to reach clean state (per D-04)
$CLI uninstall --purge --yes 2>/dev/null || true
# Verify clean: no harness-evolve in settings.json
# Verify clean: no ~/.claude/commands/evolve/ directory
# Verify clean: no ~/.harness-evolve/ directory

# Wave 1: Init & Installation
echo "=== Wave 1: Init & Installation ==="
# TEST-01: init --yes exits 0
# TEST-02: 6 hook events registered in settings.json (count harness-evolve entries using python3)
# TEST-03: scan.md exists at ~/.claude/commands/evolve/scan.md
# TEST-04: apply.md exists at ~/.claude/commands/evolve/apply.md
# TEST-05: scan.md contains template-version: 2
# TEST-06: apply.md contains template-version: 2
# TEST-07: settings.json.backup was created

# Wave 2: CLI Subcommands
echo "=== Wave 2: CLI Subcommands ==="
# TEST-08: status command exits 0
# TEST-09: scan produces valid JSON (python3 json.load succeeds)
# TEST-10: scan JSON has keys: generated_at, recommendation_count, problems, suggestions, recommendations
# TEST-11: pending produces valid JSON with "pending" key
# TEST-12: apply-one with nonexistent ID exits non-zero
# TEST-13: apply-one error output contains "not found" or success:false
# TEST-14: dismiss with nonexistent ID exits non-zero
# TEST-15: dismiss error output contains "not found" or success:false

# Wave 3: Version-Aware Re-init
echo "=== Wave 3: Version-Aware Re-init ==="
# TEST-16: Downgrade scan.md template-version to 1 (sed replace)
# TEST-17: Re-run init --yes, verify scan.md updated back to version 2
# TEST-18: Hooks still registered (not duplicated -- still 6 total)

# Wave 4: Uninstall & Teardown
echo "=== Wave 4: Uninstall & Teardown ==="
# TEST-19: uninstall --yes exits 0
# TEST-20: No harness-evolve hooks in settings.json after uninstall
# TEST-21: scan.md removed from ~/.claude/commands/evolve/
# TEST-22: apply.md removed from ~/.claude/commands/evolve/
# Re-init for purge test
$CLI init --yes >/dev/null 2>&1
# TEST-23: uninstall --purge --yes exits 0
# TEST-24: ~/.harness-evolve/ directory does not exist

# Restore original settings backup
cp "$SETTINGS.test-backup" "$SETTINGS" 2>/dev/null || true
rm -f "$SETTINGS.test-backup"

# Summary
echo ""
echo "=== RESULTS ==="
echo "PASSED: $PASS"
echo "FAILED: $FAIL"
echo "TOTAL:  $((PASS + FAIL))"
if [ "$FAIL" -gt 0 ]; then
  exit 1
fi
```
