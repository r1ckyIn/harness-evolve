    // SessionStart: fires when a Claude Code session begins, resumes, or context is cleared
    export const sessionStartInputSchema = hookCommonSchema.extend({
      hook_event_name: z.literal('SessionStart'),
      source: z.enum(['startup', 'resume', 'clear', 'compact']),
      model: z.string().optional(),

    **Step 2: Create src/hooks/session-start.ts**

    Follow the exact pattern of stop.ts. The handler should:

    1. Export `SlashCommandHealth` interface: `{ healthy: boolean; missing: string[]; outdated: string[] }`
    2. Export `RepairResult` interface: `{ repaired: boolean; details: string[] }`
    3. Export `checkAndRepairSlashCommands(homeOverride?: string)` async function:
       - Resolve dir as `join(home, '.claude', 'commands', 'evolve')` where home = homeOverride ?? process.env.HOME ?? ''
       - For each command (scan.md, apply.md):
         - Try `access(filePath)` -- if ENOENT, mark as missing
         - If exists, `readFile` and match `/<!-- template-version: (\d+) -->/` -- if no match or version < current, mark as outdated
       - If any missing or outdated: `mkdir(dir, { recursive: true })` then write files using `generateScanCommand()` / `generateApplyCommand()`
       - Return RepairResult with details array like `['scan.md installed', 'apply.md updated (v4 -> v5)']`
    4. Export `handleSessionStart(rawJson: string)` async function:
       - Parse input with `sessionStartInputSchema.parse(JSON.parse(rawJson))`
       - Call `checkAndRepairSlashCommands()`
       - If `result.repaired`, write JSON to stdout: `{ hookSpecificOutput: { hookEventName: 'SessionStart', additionalContext: '[harness-evolve] Auto-repaired ${result.details.join(', ')}. Slash commands are now up to date.' } }`
       - If not repaired, output nothing (zero latency impact for healthy sessions)
       - Wrap everything in try/catch that swallows errors (never block Claude Code)
    5. Add `main()` function that reads stdin via `readStdin()`, calls `handleSessionStart()`, then `process.exit(0)`

    Import `generateScanCommand, getScanTemplateVersion` from `../commands/evolve-scan.js`
    Import `generateApplyCommand, getApplyTemplateVersion` from `../commands/evolve-apply.js`
    Import `sessionStartInputSchema` from `../schemas/hook-input.js`
    Import `readStdin` from `./shared.js`
    Import `access, readFile, writeFile, mkdir` from `node:fs/promises`
    Import `join` from `node:path`
